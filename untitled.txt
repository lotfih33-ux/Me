function lire(k){
try{return JSON.parse(localStorage.getItem(k)||'[]')}
catch(e){return []}
}

let produits=lire(K.produits);

function sauvegarder(){
localStorage.setItem(K.clients,JSON.stringify(clients));

}

function montant(n){
return (Number(n)||0).toLocaleString('fr-FR',{
minimumFractionDigits:2,
maximumFractionDigits:2
})+' DA';
}

function safe(s){
return String(s??'').replace(/[&<>"']/g,c=>({
'&':'&amp;','<':'&lt;','>':'&gt;',
'"':'&quot;',"'":'&#39;'
}[c]));
}

function tab(id,btn){
document.querySelectorAll('.panel').forEach(x=>x.classList.remove('active'));
$(id).classList.add('active');
document.querySelectorAll('nav button').forEach(x=>x.classList.remove('active'));
if(btn)btn.classList.add('active');
}

function totaux(){
let ht=0,vat={0:0,9:0,19:0};
let utilise={0:false,9:false,19:false};

lignes.forEach(l=>{
let base=(Number(l.qte)||0)*(Number(l.prix)||0);
ht+=base;

if(l.tva!=='EXO'){
let r=Number(l.tva)||0;
if([0,9,19].includes(r)){
vat[r]+=base*r/100;
utilise[r]=true;
}
}
});

let totalTVA=vat[0]+vat[9]+vat[19];

return {ht,vat,utilise,totalTVA,ttc:ht+totalTVA};
}

function droitTimbre(a){
a=Number(a)||0;
if(a<=300)return 0;
let taux=a<=30000?1:a<=100000?1.5:2;
return Math.max(5,Math.ceil(a/100)*taux);
}

function calculer(){
let t=totaux();
let timbre=$('mode').value==='Espèces'?droitTimbre(t.ttc):0;

$('ht').textContent=montant(t.ht);

$('detailsTva').innerHTML=[0,9,19]
.filter(r=>t.utilise[r])
.map(r=>`<div><span>TVA ${r}%</span><b>${montant(t.vat[r])}</b></div>`)
.join('');

$('totalTva').textContent=montant(t.totalTVA);
$('ttc').textContent=montant(t.ttc);
$('timbre').textContent=montant(timbre);
$('net').textContent=montant(t.ttc+timbre);
}

}

function enLettres(n){
n=Math.floor(Math.abs(Number(n)||0));
if(n===0)return 'zéro';

let u=['','un','deux','trois','quatre','cinq','six','sept','huit','neuf','dix','onze','douze','treize','quatorze','quinze','seize'];

function d(x){
if(x<17)return u[x];
if(x<20)return 'dix-'+u[x-10];

let dizaines=['','','vingt','trente','quarante','cinquante','soixante','','quatre-vingt'];
let a=Math.floor(x/10),b=x%10;

if(a===7)return 'soixante-'+(b?d(10+b):'dix');
if(a===9)return 'quatre-vingt-'+(b?d(10+b):'dix');

return dizaines[a]+(b===0?(a===8?'s':''):b===1&&a<8?' et un':'-'+u[b]);
}

function c(x){
if(x<100)return d(x);
let h=Math.floor(x/100),r=x%100;
return (h===1?'cent':u[h]+' cent')+(r?' '+d(r):h>1?'s':'');
}

let parts=[];
let m=Math.floor(n/1000000);n%=1000000;
let k=Math.floor(n/1000);n%=1000;

if(m)parts.push((m===1?'un':c(m))+' million'+(m>1?'s':''));
if(k)parts.push(k===1?'mille':c(k)+' mille');
if(n)parts.push(c(n));

return parts.join(' ');
}

function exporter(){
let data={clients,produits,factures};
let blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'});
let url=URL.createObjectURL(blob);
let a=document.createElement('a');
a.href=url;
a.download='HMD-sauvegarde.json';
a.click();
URL.revokeObjectURL(url);
}

$('fichier').onchange=async function(){
let file=this.files[0];
if(!file)return;

try{
let d=JSON.parse(await file.text());

if(!Array.isArray(d.clients)||
!Array.isArray(d.produits)||
!Array.isArray(d.factures))
throw Error();

if(!confirm('Remplacer les données actuelles ?'))return;

clients=d.clients;
produits=d.produits;
factures=d.factures;

sauvegarder();
actualiserClients();
actualiserProduits();
afficherHistorique();

alert('Sauvegarde restaurée.');
}catch(e){
alert('Fichier invalide.');
}
this.value='';
};
